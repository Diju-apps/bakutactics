        <script>
            document.addEventListener("DOMContentLoaded", () => {
                const container = document.getElementById("cards-container");

                function sortCardsAlphabeticallyAndNumericallyOnLoad() {
                    const cardsEls = Array.from(container.querySelectorAll(".card"));
                    cardsEls.sort((a, b) => {
                        const nameA = a.dataset.name.toLowerCase();
                        const nameB = b.dataset.name.toLowerCase();
                        const numberA = parseInt(nameA.match(/\d+/)?.[0]) || 0;
                        const numberB = parseInt(nameB.match(/\d+/)?.[0]) || 0;
                        if (nameA.replace(/\d+/g, '') === nameB.replace(/\d+/g, '')) {
                            return numberA - numberB;
                        }
                        return nameA.localeCompare(nameB);
                    });
                    cardsEls.forEach(card => container.appendChild(card));
                    window.originalCardOrder = [...cardsEls];
                }

                fetch("/json/cards.json")
                    .then(res => res.json())
                    .then(cards => {
                        container.innerHTML = "";

                        cards.forEach(card => {
                            const type = card.type || "";
                            const isNAC = ["nac", "unac"].includes(type);
                            const isSAC = ["sac", "usac"].includes(type);
                            const isGC = ["agc", "cogc", "chgc"].includes(type);
                            const isCHGC = type === "chgc";

                            let zoomImage = "", modalImage = "", leftImage = "", rightImage = "";
                            const middleImage = card.middleImage || "";

                            if (isNAC || isSAC) {
                                const baseName = card.name.toLowerCase().replace(/[\s,:]+/g, "-");
                                zoomImage = `/images/${type}/${baseName}.png`;
                                modalImage = `/images/${type}/${type}-blank.png`;
                                leftImage = `/images/${type}/${type}-left.png`;
                                rightImage = `/images/${type}/${type}-right.png`;
                            } else if (isGC) {
                                modalImage = type === "chgc"
                                    ? `/images/gc/chgc/${card.name.toLowerCase().replace(/\s+/g, "-").replace(/-character$/, "")}.png`
                                    : `/images/gc/${type}.png`;
                                zoomImage = modalImage;
                            }

                            const finalZoomImage = zoomImage || (isNAC ? "/images/nac/nac.png" : isSAC ? "/images/sac/sac.png" : "/images/cardback.png");

                            const div = document.createElement("div");
                            div.classList.add("card", type);
                            div.tabIndex = 0;
                            div.dataset.name = card.name || "";
                            div.dataset.description = card.description || "";
                            div.dataset.type = type;
                            let rawAttr = card.attribute;
                            if (!rawAttr) {
                                const nameLower = card.name.toLowerCase();

                                if (/triple node|diagonal link|correlation|:/.test(nameLower)) {
                                    const knownAttributes = ["Pyrus", "Aquos", "Haos", "Ventus", "Darkus", "Subterra"];
                                    const matched = knownAttributes.filter(attr => nameLower.includes(attr.toLowerCase()));
                                    rawAttr = matched.join(", ");
                                }
                            }
                            rawAttr = rawAttr || "dextra";
                            const attrList = rawAttr.toLowerCase().split(/,\s*| & | and /).map(s => s.trim());
                            div.dataset.attribute = attrList.join(",");
                            div.dataset.image = finalZoomImage;
                            div.dataset.modalImage = modalImage;
                            div.dataset.leftImage = leftImage;
                            div.dataset.rightImage = rightImage;
                            div.dataset.middleImage = middleImage;
                            div.dataset.icon = card.icon || "";

                            if (["nac", "sac", "agc"].includes(type)) {
                                const attributeClass = (card.attribute || "dextra").toLowerCase();
                                div.classList.add(`attr-${attributeClass}`);
                            }

                            if (card.name.includes("Aquos & Ventus")) {
                                div.classList.add("dual-aquos-ventus");
                            } else if (card.name.includes("Darkus & Aquos")) {
                                div.classList.add("dual-darkus-aquos");
                            } else if (card.name.includes("Haos & Darkus")) {
                                div.classList.add("dual-haos-darkus");
                            } else if (card.name.includes("Pyrus & Subterra")) {
                                div.classList.add("dual-pyrus-subterra");
                            } else if (card.name.includes("Subterra & Haos")) {
                                div.classList.add("dual-subterra-haos");
                            } else if (card.name.includes("Ventus & Pyrus")) {
                                div.classList.add("dual-ventus-pyrus");
                            } else if (card.name.includes("Aquos & Subterra")) {
                                div.classList.add("dual-aquos-subterra");
                            } else if (card.name.includes("Haos & Ventus")) {
                                div.classList.add("dual-haos-ventus");
                            } else if (card.name.includes("Pyrus & Darkus")) {
                                div.classList.add("dual-pyrus-darkus");
                            }

                            const iconPath = card.attribute
                                ? `/images/icon/${card.attribute.toLowerCase()}.png`
                                : `/images/icon/dextra.png`;

                            if (isNAC || isSAC || isGC) {
                                const mask = document.createElement("div");
                                mask.className = "mask";
                                const stack = document.createElement("div");
                                stack.className = "image-stack";

                                if (isNAC || isSAC) {
                                    [["base", modalImage, 3], ["right", rightImage, 2], ["left", leftImage, 2], ["top", finalZoomImage, 1]].forEach(([alt, src, z]) => {
                                        const img = document.createElement("img");
                                        img.src = src;
                                        img.alt = `${card.name} ${alt}`;
                                        img.style.zIndex = z;
                                        stack.appendChild(img);
                                    });
                                } else {
                                    const img = document.createElement("img");
                                    img.src = modalImage;
                                    img.alt = card.name;
                                    img.style.zIndex = 3;
                                    img.setAttribute("draggable", "false");
                                    stack.appendChild(img);
                                }

                                mask.appendChild(stack);
                                div.appendChild(mask);
                            }

                            if (isGC || isNAC || isSAC) {
                                const nameEl = document.createElement("h2");
                                nameEl.className = "card-title";
                                nameEl.innerHTML = `<img src="${iconPath}" alt="${card.attribute || 'dextra'}" class="attribute-icon" /> ${card.name}`;
                                div.appendChild(nameEl);
                            }

                            let highlightedDescription = window.highlightKeywords(card.description || "");
                            highlightedDescription = window.highlightAttributes(highlightedDescription, isGC);
                            highlightedDescription = isSAC
                                ? window.highlightBakuganNames(highlightedDescription)
                                : isCHGC
                                ? window.highlightBakuganNames(highlightedDescription, true)
                                : highlightedDescription;

                            const p = document.createElement("p");
                            p.innerHTML = highlightedDescription;
                            div.appendChild(p);

                            const hspContainer = document.createElement("div");
                            hspContainer.className = "hsp-container";
                            hspContainer.innerHTML = `
                                <span class="hsp-value">${card.hsp || ""}</span>
                                <span class="number" data-number="${card.limit || 0}">${card.limit || 0}</span>
                            `;
                            div.appendChild(hspContainer);

                            container.appendChild(div);
                        });

                        sortCardsAlphabeticallyAndNumericallyOnLoad();
                    })
                    .catch(err => console.error("Failed to load cards:", err));
                });
        </script>
    </body>
